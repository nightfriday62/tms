create table IF NOT EXISTS T_TRUCK
(
    PLATE_NUMBER         varchar(15) primary key,
    TRUCK_TYPE           ENUM('RIGID', 'SEMI'),
    BRAND                varchar(15),
    MODEL                varchar(15),
    VOLUME               decimal(8, 2),
    WEIGHT               decimal(8, 2),
    TRAILER_TYPE         ENUM('FLATBED', 'BOX', 'REFRIGERATED','OTHER'),
    TRAILER_PLATE_NUMBER varchar(15)
) ;

create table IF NOT EXISTS T_SHIPMENT
(
    SHIPMENT_NUMBER varchar(15) primary key,
    ORIGIN          varchar(50),
    DESTINATION     varchar(50),
    WEIGHT          decimal(8, 2),
    VOLUME          decimal(8, 2),
    STATUS          ENUM('SCHEDULED', 'SHIPPED', 'DELIVERED'),
    SCHEDULED_SHIPMENT_DATE date,
    ACTUAL_SHIPMENT_DATE    date,
    SCHEDULED_DELIVERY_DATE date,
    ACTUAL_DELIVERY_DATE    date
);

CREATE TABLE IF NOT EXISTS SHIPPING_UNIT (
    UNIT_ID INT NOT NULL PRIMARY KEY AUTO_INCREMENT,
    DESCRIPTION VARCHAR(200),
    WEIGHT DECIMAL(8, 2),
    HEIGHT DECIMAL(8, 2),
    WIDTH DECIMAL(8, 2),
    `LENGTH` DECIMAL(8, 2),
    VOLUME DECIMAL(8, 2)
);

CREATE TABLE IF NOT EXISTS SHIPMENT_TRUCK (
    `ID` INT NOT NULL PRIMARY KEY AUTO_INCREMENT,
    SHIPMENT_NUMBER VARCHAR(15),
    PLATE_NUMBER VARCHAR(15),
    FOREIGN KEY (SHIPMENT_NUMBER) REFERENCES T_SHIPMENT(SHIPMENT_NUMBER),
    FOREIGN KEY (PLATE_NUMBER) REFERENCES T_TRUCK(PLATE_NUMBER)
);

CREATE TABLE IF NOT EXISTS UNIT_SHIPMENT (
    ID INT NOT NULL PRIMARY KEY AUTO_INCREMENT,
    UNIT_ID INT,
    SHIPMENT_NUMBER VARCHAR(15),
    FOREIGN KEY (UNIT_ID) REFERENCES SHIPPING_UNIT(UNIT_ID),
    FOREIGN KEY (SHIPMENT_NUMBER) REFERENCES T_SHIPMENT(SHIPMENT_NUMBER)
);

CREATE TABLE IF NOT EXISTS SHIPMENT_SEQUENCE (
    ID INT NOT NULL PRIMARY KEY AUTO_INCREMENT,
    `DATE` DATE,
    `SEQUENCE` INT
);
